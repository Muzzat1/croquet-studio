const PrecisionWheel = ({ value, onChange, isPlaying, min = 0, max = 360, sensitivity = 0.2, range = 60, wrap = true, unit = "°", label = "Wheel", showLabels = false, lightMode = false }: any) => {
  const [isDragging, setIsDragging] = useState(false); const lastPosRef = useRef<number | null>(null); const containerRef = useRef<HTMLDivElement>(null);
  const handleStart = (e: React.MouseEvent | React.TouchEvent) => { if (isPlaying) return; setIsDragging(true); lastPosRef.current = 'touches' in e ? e.touches[0].clientX : (e as React.MouseEvent).clientX; };
  useEffect(() => {
    const handleMove = (e: MouseEvent | TouchEvent) => {
      if (!isDragging || isPlaying || lastPosRef.current === null) return;
      const currentPos = 'touches' in e ? e.touches[0].clientX : (e as MouseEvent).clientX; const delta = lastPosRef.current - currentPos;
      let newValue = value + delta * sensitivity;
      if (wrap) { const span = max - min; newValue = ((newValue - min) % span + span) % span + min; } else { newValue = Math.max(min, Math.min(max, newValue)); }
      onChange(newValue); lastPosRef.current = currentPos;
    };
    const handleEnd = () => { setIsDragging(false); lastPosRef.current = null; };
    if (isDragging) { window.addEventListener('mousemove', handleMove); window.addEventListener('mouseup', handleEnd); window.addEventListener('touchmove', handleMove); window.addEventListener('touchend', handleEnd); }
    return () => { window.removeEventListener('mousemove', handleMove); window.removeEventListener('mouseup', handleEnd); window.removeEventListener('touchmove', handleMove); window.removeEventListener('touchend', handleEnd); };
  }, [isDragging, value, onChange, isPlaying, min, max, sensitivity, wrap]);

  const ticks = []; const startVal = value - range / 2; const endVal = value + range / 2;
  for (let i = Math.floor(startVal); i <= Math.ceil(endVal); i++) {
    const span = max - min; const displayVal = wrap ? ((i - min) % span + span) % span + min : i; const pos = ((i - startVal) / range) * 100;
    if (pos >= -5 && pos <= 105 && (!wrap ? (i >= min && i <= max) : true)) { ticks.push({ val: displayVal, pos }); }
  }
  const getTickLabel = (val: number) => {
    if (!showLabels) return `${Math.round(val)}`; const v = Math.round(val);
    if (v === 0 || v === 360) return 'N'; if (v === 90) return 'E'; if (v === 180) return 'S'; if (v === 270) return 'W'; return `${v}`;
  };

  return (
      <div className="flex flex-col gap-1 w-full">
        <div className="flex justify-between items-end">
           <span className={`text-[9px] font-bold uppercase tracking-widest ${lightMode ? 'text-zinc-600' : 'text-zinc-400'}`}>{label}</span>
           <div className="flex items-center gap-0.5">
              <button onMouseDown={(e) => { e.stopPropagation(); let n = value - (wrap ? 1 : 5); if(wrap) n = (n+360)%360; else n = Math.max(min, n); onChange(n); }} className={`rounded cursor-pointer p-0.5 border shadow-sm transition-colors ${lightMode ? 'text-zinc-600 hover:text-emerald-700 bg-white/60 border-white/80 hover:bg-white/90' : 'text-zinc-500 hover:text-emerald-400 bg-zinc-900 border-zinc-800'}`}><ChevronLeft size={10}/></button>
              <span className={`font-mono text-[10px] font-bold px-1 py-0.5 rounded border shadow-sm w-[46px] text-center ${lightMode ? 'text-emerald-700 bg-white/80 border-white/90' : 'text-emerald-400 bg-zinc-950/80 border-zinc-700'}`}>{value.toFixed(wrap ? 1 : 0)}{unit}</span>
              <button onMouseDown={(e) => { e.stopPropagation(); let n = value + (wrap ? 1 : 5); if(wrap) n = (n)%360; else n = Math.min(max, n); onChange(n); }} className={`rounded cursor-pointer p-0.5 border shadow-sm transition-colors ${lightMode ? 'text-zinc-600 hover:text-emerald-700 bg-white/60 border-white/80 hover:bg-white/90' : 'text-zinc-500 hover:text-emerald-400 bg-zinc-900 border-zinc-800'}`}><ChevronRight size={10}/></button>
           </div>
        </div>
        <div ref={containerRef} onMouseDown={handleStart} onTouchStart={handleStart} className={`relative rounded-lg overflow-hidden select-none touch-none transition-all shadow-inner h-8 w-full border ${isPlaying ? 'opacity-50' : 'cursor-pointer hover:border-emerald-500/50'} ${lightMode ? 'bg-white/50 border-white/60 shadow-[inset_0_2px_4px_rgba(0,0,0,0.05)]' : 'bg-zinc-950/80 border-zinc-700'}`}>
          <div className="absolute inset-0 flex items-center">
           {ticks.map((tick, idx) => (
            <div key={idx} className="absolute flex flex-col items-center justify-center" style={{ left: `${tick.pos}%`, transform: 'translateX(-50%)' }}>
              <div className={`w-px ${tick.val % 10 === 0 ? (lightMode ? 'h-3 bg-zinc-400' : 'h-3 bg-zinc-300') : tick.val % 5 === 0 ? (lightMode ? 'h-2 bg-zinc-300' : 'h-2 bg-zinc-500') : (lightMode ? 'h-1 bg-zinc-300/50' : 'h-1 bg-zinc-700')}`} />
              {tick.val % 10 === 0 && ( <span className={`text-[8px] font-bold font-mono mt-1 absolute whitespace-nowrap ${lightMode ? 'text-zinc-600' : 'text-zinc-400'}`}>{getTickLabel(tick.val)}</span> )}
            </div>
          ))}
        </div>
        <div className="absolute inset-0 flex justify-center items-center pointer-events-none">
          <div className="w-0.5 h-full bg-emerald-500 relative z-10 shadow-[0_0_8px_rgba(16,185,129,0.8)]"><div className="absolute bg-emerald-500 top-0 left-1/2 -translate-x-1/2 w-1.5 h-1 rounded-b-sm" /><div className="absolute bg-emerald-500 bottom-0 left-1/2 -translate-x-1/2 w-1.5 h-1 rounded-t-sm" /></div>
        </div>
        <div className={`absolute inset-0 pointer-events-none z-0 bg-gradient-to-r ${lightMode ? 'from-white/40 via-transparent to-white/40' : 'from-zinc-950 via-transparent to-zinc-950'}`} />
      </div>
    </div>
  );
};

const ToolButton = ({ icon, label, active, onClick, danger, lightMode }: any) => {
  const baseClasses = "flex flex-col items-center justify-center gap-1.5 p-2 min-w-[64px] rounded-xl border-2 transition-all duration-200 shadow-sm active:scale-95 cursor-pointer";
  let stateClasses = "";
  if (active) {
    stateClasses = lightMode 
      ? "bg-emerald-50 text-emerald-700 border-emerald-500 shadow-[0_4px_15px_rgba(16,185,129,0.25)] hover:bg-emerald-100" 
      : "bg-emerald-500/10 text-emerald-400 border-emerald-500 shadow-[0_4px_15px_rgba(16,185,129,0.2)] hover:bg-emerald-500/20";
  } else if (danger) {
    stateClasses = lightMode
      ? "bg-white text-rose-600 border-rose-200 hover:border-rose-400 hover:bg-rose-50 hover:shadow-md hover:-translate-y-0.5"
      : "bg-slate-800/80 text-rose-400 border-slate-700 hover:border-rose-500/50 hover:bg-rose-500/10 hover:-translate-y-0.5";
  } else {
    stateClasses = lightMode
      ? "bg-white text-slate-600 border-slate-200 hover:border-emerald-300 hover:text-emerald-600 hover:shadow-md hover:-translate-y-0.5"
      : "bg-slate-800/80 text-slate-400 border-slate-700 hover:border-emerald-500/50 hover:text-emerald-300 hover:shadow-lg hover:-translate-y-0.5";
  }

  return (
    <button onClick={onClick} title={label} className={`${baseClasses} ${stateClasses}`}>
      {icon}
      <span className="text-[9px] font-bold uppercase tracking-widest">{label}</span>
    </button>
  );
};

            }
        }

        </main>

        {cleanFeed && (
          <div className="absolute bottom-4 right-4 z-[200] flex gap-2 pointer-events-auto">
            <button onClick={() => {
              const url = canvasRef.current?.toDataURL('image/png');
              if (url) {
                const a = document.createElement('a'); a.href = url;
                const d = new Date();
                a.download = `croquet_capture_${d.getFullYear()}${(d.getMonth()+1).toString().padStart(2, '0')}${d.getDate().toString().padStart(2, '0')}_${d.getHours().toString().padStart(2, '0')}${d.getMinutes().toString().padStart(2, '0')}.png`;
                a.click();
              }
            }} className="bg-emerald-600/90 hover:bg-emerald-500 text-white px-4 py-2 rounded-lg text-[10px] font-bold uppercase tracking-wider border border-emerald-500 transition-all shadow-lg flex items-center gap-2">
              <Camera size={14} /> Capture Screen
            </button>
            <button onClick={() => setCleanFeed(false)} className={`px-4 py-2 rounded-lg text-[10px] font-bold uppercase tracking-wider border transition-all shadow-lg backdrop-blur-md ${lightMode ? 'bg-white/90 hover:bg-white text-zinc-800 border-white/50' : 'bg-zinc-900/80 hover:bg-zinc-800 text-zinc-300 hover:text-white border-zinc-700'}`}>
              Exit Freeze Frame
            </button>
          </div>
        )}

        {/* === UPDATED HELP MENU === */}
        {showHelp && !cleanFeed && (
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className={`absolute inset-0 z-[150] backdrop-blur-md flex items-center justify-center p-4 ${lightMode ? 'bg-white/30' : 'bg-zinc-950/90'}`}>
            <motion.div initial={{ scale: 0.9, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} className={`rounded-2xl w-full max-w-4xl shadow-2xl border backdrop-blur-xl ${lightMode ? 'bg-white/90 border-white/50' : 'bg-zinc-900 border-zinc-800'}`}>
              <div className={`p-6 border-b flex items-center justify-between rounded-t-2xl ${lightMode ? 'bg-white/50 border-white/50' : 'bg-zinc-900 border-zinc-800'}`}>
                <h2 className={`text-2xl font-bold flex items-center gap-3 ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><HelpCircle size={28} /> Studio Guide</h2>
                <button onClick={() => setShowHelp(false)} className={`p-2 rounded-full transition-colors ${lightMode ? 'hover:bg-white/80 text-zinc-500 hover:text-zinc-800' : 'hover:bg-zinc-800 text-zinc-400 hover:text-white'}`}><X size={20} /></button>
              </div>
              
              <div className={`p-8 grid grid-cols-1 md:grid-cols-2 gap-x-12 gap-y-8 text-sm leading-relaxed ${lightMode ? 'text-zinc-700' : 'text-zinc-300'}`}>
                <div className="space-y-6">
                    <div>
                        <h3 className={`font-bold mb-3 flex items-center gap-2 uppercase tracking-widest text-[11px] ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><Hand size={16}/> Getting Started</h3>
                        <p className="mb-2"><strong>Spawn Balls:</strong> Click and drag balls from the Caddy on the right onto the green. They will scale down automatically.</p>
                        <p><strong>Selection:</strong> Click any ball to make it active. Click the empty grass to clear your selection.</p>
                    </div>

                    <div>
                        <h3 className={`font-bold mb-3 flex items-center gap-2 uppercase tracking-widest text-[11px] ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><Play size={16}/> Aiming & Shooting</h3>
                        <p className="mb-3 flex items-start gap-2"><Eye size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>AIM Mode:</strong> Use the precision wheels to adjust Velocity and Bearing. The ghost ball shows your impact point.</span></p>
                        <p className="mb-3 flex items-start gap-2"><MousePointer2 size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>PLACE Mode:</strong> Select a ball, then click anywhere on the lawn. The app calculates the exact angle and power for you.</span></p>
                        <p className="flex items-start gap-2"><Undo2 size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>Undo & Reset:</strong> Use the <RotateCcw size={14} className="inline mx-1"/> button to clear the court, or the <Undo2 size={14} className="inline mx-1"/> button to step backwards if you make a mistake.</span></p>
                    </div>
                </div>

                <div className="space-y-6">
                    <div>
                        <h3 className={`font-bold mb-3 flex items-center gap-2 uppercase tracking-widest text-[11px] ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><Pencil size={16}/> Tactics & Drawing</h3>
                        <p className="mb-2 flex items-start gap-2"><Pencil size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>Telestrator:</strong> Click the Draw tool to sketch tactical lines directly onto the grass.</span></p>
                        <p className="flex items-start gap-2"><Eraser size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>Clear Ink:</strong> Use the Eraser to instantly remove all drawings.</span></p>
                    </div>

                    <div>
                        <h3 className={`font-bold mb-3 flex items-center gap-2 uppercase tracking-widest text-[11px] ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><Clapperboard size={16}/> Broadcast Tools</h3>
                        <p className="mb-3 flex items-start gap-2"><Sun size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>Outdoor Overlay:</strong> Enable in <Settings size={14} className="inline mx-1 text-zinc-400"/> Prefs. Switches to a premium frosted glass UI so you can read the app outside in the sun.</span></p>
                        <p className="mb-3 flex items-start gap-2"><MonitorPlay size={16} className={`shrink-0 mt-0.5 ${lightMode ? 'text-zinc-400' : 'text-zinc-500'}`} /> <span><strong>Freeze Frame:</strong> Enable in Prefs. Hides all menus for recording. You can also capture a PNG screenshot from here.</span></p>
                        <p><strong>Sequence Record:</strong> Capture "Frames" with the camera icon to animate complex tactical breaks.</p>
                    </div>
                </div>

              </div>
              
              <div className={`p-6 border-t rounded-b-2xl text-center ${lightMode ? 'bg-white/50 border-white/50' : 'bg-zinc-950/50 border-zinc-800'}`}>
                <button onClick={() => setShowHelp(false)} className="px-10 py-3 bg-emerald-500 hover:bg-emerald-400 text-zinc-950 rounded-xl font-bold uppercase tracking-widest text-[11px] transition-colors shadow-lg">Close Guide</button>
              </div>
            </motion.div>
          </motion.div>
        )}

        <AnimatePresence>
          {showOptions && !cleanFeed && (
            <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className={`absolute inset-0 z-[150] backdrop-blur-md flex items-center justify-center p-4 ${lightMode ? 'bg-white/30' : 'bg-zinc-950/90'}`}>
              <motion.div initial={{ scale: 0.9, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} className={`rounded-2xl w-full max-w-md shadow-2xl border backdrop-blur-xl ${lightMode ? 'bg-white/90 border-white/50' : 'bg-zinc-900 border-zinc-800'}`}>
                <div className={`p-6 border-b flex items-center justify-between sticky top-0 z-10 rounded-t-2xl ${lightMode ? 'bg-white/50 border-white/50' : 'bg-zinc-900 border-zinc-800'}`}>
                  <h2 className={`text-xl font-bold flex items-center gap-2 ${lightMode ? 'text-emerald-700' : 'text-emerald-400'}`}><Settings size={24} />Studio Options</h2>
                  <button onClick={() => setShowOptions(false)} className={`p-2 rounded-full transition-colors ${lightMode ? 'hover:bg-white/80 text-zinc-500 hover:text-zinc-800' : 'hover:bg-zinc-800 text-zinc-400 hover:text-white'}`}><X size={20} /></button>
                </div>
                <div className="p-6 space-y-3">
                  
                  <div className={`flex items-center justify-between p-4 rounded-xl border shadow-sm ${lightMode ? 'bg-white/60 border-white/60' : 'bg-zinc-950 border-zinc-800'}`}>
                     <div className={`flex items-center gap-3 font-bold text-sm uppercase tracking-wider ${lightMode ? 'text-zinc-700' : 'text-zinc-300'}`}><Sun size={16} /> Outdoor Mode</div>
                     <button onClick={() => setLightMode(!lightMode)} className={`w-12 h-6 rounded-full relative transition-colors shadow-inner border ${lightMode ? 'bg-emerald-500 border-emerald-600' : 'bg-zinc-700 border-zinc-900'}`}>
                       <div className={`absolute top-1 w-4 h-4 rounded-full bg-white transition-all shadow-md ${lightMode ? 'left-7' : 'left-1'}`} />
                     </button>
                  </div>

                  <div className={`flex items-center justify-between p-4 rounded-xl border shadow-sm ${lightMode ? 'bg-white/60 border-white/60' : 'bg-zinc-950 border-zinc-800'}`}>
                     <div className={`flex items-center gap-3 font-bold text-sm uppercase tracking-wider ${lightMode ? 'text-zinc-700' : 'text-zinc-300'}`}><Save size={16} /> File Prefix</div>
                     <input type="text" value={filenamePrefix} onChange={(e) => setFilenamePrefix(e.target.value)} className={`text-xs font-bold px-3 py-1.5 rounded outline-none focus:ring-1 focus:ring-emerald-500 w-32 text-right select-auto border shadow-inner ${lightMode ? 'bg-white/80 border-white/90 text-zinc-800' : 'bg-zinc-900 border-zinc-700 text-zinc-300'}`} style={{ WebkitTouchCallout: 'default', WebkitUserSelect: 'auto', userSelect: 'auto' }} placeholder="sequence" />
                  </div>

                  <div className={`flex items-center justify-between p-4 rounded-xl border shadow-sm ${lightMode ? 'bg-white/60 border-white/60' : 'bg-zinc-950 border-zinc-800'}`}>
                     <div className={`flex items-center gap-3 font-bold text-sm uppercase tracking-wider ${lightMode ? 'text-zinc-700' : 'text-zinc-300'}`}><MonitorPlay size={16} /> Freeze Frame Mode</div>
                     <button onClick={() => { setCleanFeed(true); setShowOptions(false); }} className={`px-4 py-1.5 rounded text-[11px] font-bold uppercase tracking-widest border transition-colors shadow-sm ${lightMode ? 'bg-emerald-500/20 text-emerald-700 border-emerald-500/50 hover:bg-emerald-500/30' : 'bg-emerald-900/40 text-emerald-400 border-emerald-500/50 hover:bg-emerald-800/60'}`}>Activate</button>
                  </div>

                  {[ { key: 'recording', label: 'Sequence Record', icon: <Clapperboard size={16} /> }, { key: 'zoom', label: 'Pan & Zoom Controls', icon: <ZoomIn size={16} /> } ].map(f => (
                    <div key={f.key} className={`flex items-center justify-between p-4 rounded-xl border shadow-sm ${lightMode ? 'bg-white/60 border-white/60' : 'bg-zinc-950 border-zinc-800'}`}>
                       <div className={`flex items-center gap-3 font-bold text-sm uppercase tracking-wider ${lightMode ? 'text-zinc-700' : 'text-zinc-300'}`}>{f.icon} {f.label}</div>
                       <button onClick={() => { setFeatures(prev => { const newState = !prev[f.key as keyof typeof features]; if (f.key === 'zoom' && !newState) { setZoom(1); setTimeout(() => { if (viewportRef.current) { viewportRef.current.scrollLeft = (viewportRef.current.scrollWidth - viewportRef.current.clientWidth) / 2; viewportRef.current.scrollTop = viewportRef.current.scrollHeight; } }, 10); } return { ...prev, [f.key]: newState }; }); }} className={`w-12 h-6 rounded-full relative transition-colors shadow-inner border ${features[f.key as keyof typeof features] ? (lightMode ? 'bg-emerald-500 border-emerald-600' : 'bg-emerald-500 border-emerald-700') : (lightMode ? 'bg-white/80 border-zinc-300' : 'bg-zinc-700 border-zinc-900')}`}>
                         <div className={`absolute top-1 w-4 h-4 rounded-full bg-white transition-all shadow-md ${features[f.key as keyof typeof features] ? 'left-7' : 'left-1'}`} />
                       </button>
                    </div>
                  ))}
                </div>
              </motion.div>
            </motion.div>
          )}
        </AnimatePresence>
      </div>
    </>
  );
}
